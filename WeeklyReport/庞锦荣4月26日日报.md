### 题目1：[70.爬楼梯](https://leetcode.cn/problems/climbing-stairs/)

![屏幕截图 2026-04-26 224111](C:\Users\dyyxs\Desktop\截图\屏幕截图 2026-04-26 224111.png)

```cpp
class Solution {
public:
    int ret;
    void Back_Tracking(int n, int sum) {
        if (sum >= n) {
            if (sum == n) {
                ret++;
            }
            return;
        }
        sum += 1;
        Back_Tracking(n, sum);
        sum -= 1;
        sum += 2;
        Back_Tracking(n, sum);
        sum -= 2;
    }
    int climbStairs(int n) {
        if (1 == n || 2 == n) {
            return n;
        }
        Back_Tracking(n, 0);
        return ret;
    }
};
```
